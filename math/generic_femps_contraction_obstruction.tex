\section{Direct one-form Cayley coefficient}

Let $A=(A_{ij})$ be an $n\times n$ matrix over $\Mat_d(\mathbb F)$ and define
the site-dependent matrix-valued one-forms
\[
  \mathcal A^{[i]}=\sum_{j=1}^n A_{ij}e_j,
  \qquad 1\le i\le n.
\]
For boundary vectors $u,v\in\mathbb F^d$, set
\[
  \Phi(A;u,v)=u^{\mathsf T}
  \mathcal A^{[1]}\wedge\cdots\wedge\mathcal A^{[n]}v
  \in\Lambda^n\mathbb F^n.
\]
After absorbing $u$ and $v$ into the endpoint cores, this is an open-boundary
one-form FEMPS whose internal bonds are all at most $d$.

\begin{lemma}[Direct coefficient identity]
With the increasing top form given unit norm,
\[
  \Phi(A;u,v)=
  \bigl(u^{\mathsf T}\CayleyDet(A)v\bigr)
  e_1\wedge\cdots\wedge e_n.
\]
\end{lemma}

\begin{proof}
Expanding the wedge, a nonzero top-degree term chooses every basis vector once
and is therefore indexed by a permutation $\sigma\in S_n$.  Its exterior sign
is $\operatorname{sgn}(\sigma)$, while its virtual contraction preserves the
site order and equals
$u^{\mathsf T}A_{1,\sigma(1)}\cdots A_{n,\sigma(n)}v$.
Summing the terms gives the row-ordered Cayley determinant.
\end{proof}

\begin{theorem}[Fixed-bond exact squared-norm obstruction]
Over $\mathbb Q$, exact squared-norm evaluation for rational one-form
matrix-wedge FEMPS is \#P-hard under polynomial-time Turing reductions even
when $D=N$ and every internal bond is at most three.  Thus a generic exact
contraction algorithm polynomial in $(N,D)$ for the fixed class $\chi\le3$
would imply $\mathrm{FP}=\#\mathrm P$.
\end{theorem}

\begin{proof}
Computing $\CayleyDet(A)$ for
$A\in\Mat_n(\Mat_2(\mathbb Q))$ is \#P-hard
\cite{ChienEtAl2011}.  For each pair of standard boundary vectors $u,v$, the
lemma gives a bond-two state with top coefficient
$x=u^{\mathsf T}\CayleyDet(A)v$ and squared norm $x^2$.

Let $\Theta=e_1\wedge\cdots\wedge e_n$.  A block-diagonal virtual direct sum of
the bond-two hard cores and the bond-one cores for $\Theta$ represents
$\Phi+\Theta$ with maximum bond three and squared norm $(x+1)^2$.  Two exact
squared-norm queries recover the signed rational amplitude through
\[
  x=\frac{(x+1)^2-x^2-1}{2}.
\]
The four standard boundary pairs recover the matrix output.  The direct-sum
cores, their rational bit lengths, and the factorial-free normalization above
have descriptions polynomial in the source input.  The resulting states are
alternating by construction, so their antisymmetry residual is exactly zero.
\end{proof}

This theorem concerns site-dependent one-form cores.  It does not contradict
the polynomial LC--AGP collapse of a fixed $\Mat_2$ \emph{homogeneous pair
power}: repeating one even two-form symmetrizes factor order, whereas the
one-form construction retains the row order through the physical site label.

\section{Matrix-valued pair powers}

Let $V=\mathbb F^D$, let $\mathcal A=\Mat_\chi(\mathbb F)$, and let
$B_{pq}\in\mathcal A$ satisfy $B_{qp}=-B_{pq}$ and $B_{pp}=0$. Define
\[
  \Omega_B=\sum_{p<q}B_{pq}\,e_p\wedge e_q
  \in \mathcal A\otimes\Lambda^2V.
\]
For $N=2M$ and boundary vectors $l,r\in\mathbb F^\chi$, set
\[
  \Psi_M(B;l,r)=l^{\mathsf T}\frac{\Omega_B^M}{M!}r.
\]
Products in $\mathcal A$ retain their order. Since two-forms have even degree,
the coefficient on a support $S$ of size $2M$ is
\begin{equation}
  C_S=\frac1{M!}\sum_{\mathfrak m\in\operatorname{PM}(S)}
  \operatorname{sgn}(\mathfrak m)
  \sum_{\tau\in S_M}
  B_{a_{\tau(1)}b_{\tau(1)}}\cdots
  B_{a_{\tau(M)}b_{\tau(M)}}.
  \label{eq:sym-pf}
\end{equation}
Here $\mathfrak m=\{(a_k,b_k):a_k<b_k\}$ is a perfect matching and its sign
is the exterior reordering sign. Formula~\eqref{eq:sym-pf} is independent of
the initially chosen enumeration of matching edges because of the inner
symmetrization.

\section{Shift-tag identity}

For an $n\times n$ matrix $A=(A_{ij})$ over $\Mat_d(\mathbb F)$, define its
row-ordered Cayley determinant by
\[
  \CayleyDet(A)=\sum_{\sigma\in S_n}\operatorname{sgn}(\sigma)
  A_{1,\sigma(1)}\cdots A_{n,\sigma(n)}.
\]

\begin{lemma}[Shift-tag reduction]
Assume $n!$ is invertible in $\mathbb F$. Take $D=2n$ with physical indices
$x_1,\ldots,x_n,y_1,\ldots,y_n$, take $\chi=(n+1)d$, and identify the
virtual space with $\mathbb F^{n+1}\otimes\mathbb F^d$. Set
\[
 B_{x_i y_j}=E_{i,i+1}\otimes A_{ij},
 \qquad B_{x_ix_j}=B_{y_i y_j}=0,
\]
and extend by skew symmetry. If $l=e_1\otimes u$ and
$r=e_{n+1}\otimes v$, then the unique top-degree coefficient is
\[
  [e_{x_1}\wedge\cdots\wedge e_{x_n}\wedge
    e_{y_1}\wedge\cdots\wedge e_{y_n}]\,\Psi_n
  =\frac{(-1)^{n(n-1)/2}}{n!}
    u^{\mathsf T}\CayleyDet(A)v.
\]
\end{lemma}

\begin{proof}
Only matchings joining every $x_i$ to a distinct $y_{\sigma(i)}$ can
contribute. Relative to the displayed physical order, the sign is
\[
 (-1)^{n(n-1)/2}\operatorname{sgn}(\sigma).
\]
For any fixed
matching, Equation~\eqref{eq:sym-pf} symmetrizes all $n!$ factor orders.
The matrix-unit product is zero for every order except increasing row order,
where
\[
 E_{12}E_{23}\cdots E_{n,n+1}=E_{1,n+1}.
\]
Contracting the tag endpoints and summing over $\sigma$ gives the formula.
\end{proof}

\section{One-form FEMPS embedding}

\begin{lemma}[Polynomial-bond embedding]
Every state $\Psi_M(B;l,r)$ has an exact open-chain one-form matrix-wedge
FEMPS representation with $2M$ particle sites and bonds alternating between
$\chi D$ and $\chi$, apart from the unit boundary bonds.
\end{lemma}

\begin{proof}
For each pair factor introduce an intermediate index $(a,i)$ of size
$\chi D$. The odd core maps $a$ to $(a,i)$ while emitting $e_i$. The even
core maps $(a,i)$ to $b$ while emitting
$\sum_{j>i}(B_{ij})_{ab}e_j$. Their matrix-wedge product is exactly the
matrix-valued two-form $\Omega_B$. Repeat the two cores $M$ times, insert
$l^{\mathsf T}/M!$ in the first core, and contract $r$ in the last core.
\end{proof}

\begin{theorem}[Independent homogeneous pair-power obstruction]
Over $\mathbb Q$, exact squared-norm evaluation for rational matrix-valued
pair powers is \#P-hard under polynomial-time Turing reductions.  The
construction has $D=N$ and embeds in one-form FEMPS with maximum bond
$O(N^2)$.
\end{theorem}

\begin{proof}
By Chien--Harsha--Sinclair--Srinivasan, computing
$\CayleyDet(A)$ for $A\in\Mat_n(\Mat_2(\mathbb Q))$ is \#P-hard. Apply the
shift-tag lemma with $d=2$. Four choices of $u,v$ recover the four entries of
the Cayley determinant up to a known nonzero rational scale. The resulting
pair state has $D=N=2n$ and pair bond $2(n+1)$; the embedding lemma gives
maximum one-form FEMPS bond $4n(n+1)=O(N^2)$. Its explicit description and
all rational bit lengths are polynomial in the input size.

It remains to reduce a signed amplitude $x$ to squared norms. The top exterior space
$\Lambda^{2n}\mathbb Q^{2n}$ is one-dimensional. A scalar block with only
$B_{x_i y_i}=1$ supplies a known top coefficient
$q=(-1)^{n(n-1)/2}$. A virtual direct sum adds this scalar block to the hard
block, producing amplitude $x+q$ with only constant additive bond overhead.
Exact squared-norm queries give $x^2$ and $(x+q)^2$, hence
\[
 x=\frac{(x+q)^2-x^2-q^2}{2q}.
\]
Thus an exact polynomial-time squared-norm oracle computes every required
Cayley determinant entry and therefore solves a \#P-hard problem.
\end{proof}

\section{A simpler sparse-path obstruction}

The generic consequence above has a direct proof that does not require
noncommutative determinant hardness. It also shows that dense virtual matrices
and multiple virtual paths are not the source of the obstruction.

Let $n=M$, $D=2n$, and define orthonormal pair forms
\[
 P_j=e_{2j-1}\wedge e_{2j},\qquad 1\le j\le n.
\]
Given a zero--one matrix $A\in\{0,1\}^{n\times n}$, take virtual dimension
$n+1$ and set
\[
 B_{2j-1,2j}=\sum_{i=1}^n A_{ij}E_{i,i+1},
\]
with every other independent physical-pair coefficient zero. Equivalently,
the matrix of two-forms is upper bidiagonal with edge forms
\[
 F_i=\sum_{j=1}^n A_{ij}P_j.
\]

\begin{theorem}[Bandwidth-one permanent obstruction]
With boundaries $l=e_1$ and $r=e_{n+1}$,
\[
 \Psi_n(B;l,r)
 =\frac{\operatorname{perm}(A)}{n!}
   P_1\wedge\cdots\wedge P_n,
 \qquad
 \langle\Psi_n,\Psi_n\rangle
 =\frac{\operatorname{perm}(A)^2}{(n!)^2}.
\]
Consequently exact squared-norm evaluation is \#P-hard already for upper-
bidiagonal pair matrices with one endpoint path, $D=2n$, pair bond $n+1$,
and $O(n^2)$ binary input coefficients.
\end{theorem}

\begin{proof}
The unique length-$n$ virtual path gives
$F_1\wedge\cdots\wedge F_n/n!$. In its multilinear expansion, a repeated
$P_j$ vanishes. Every surviving term therefore selects a permutation
$\sigma\in S_n$. Pair forms have even degree and commute, so every term has
the same top form, with total coefficient
$\sum_\sigma\prod_i A_{i,\sigma(i)}=\operatorname{perm}(A)$. The top form has
unit norm. Since the permanent of a zero--one matrix is \#P-complete
\cite{Valiant1979}, an exact
squared-norm value recovers it by multiplying by $(n!)^2$ and taking the
nonnegative integer square root.
\end{proof}

The state is an antisymmetrized product of geminals; permanent-valued APG/APIG
coefficients and Fischer APG-to-AGP expansions are established
\cite{RicherKimAyers2025,KawasakiNakatani2024}. This theorem is a compact-
input contraction obstruction, not a lower bound on physical LC--AGP rank: the
output lies in the one-dimensional top exterior sector. The row-order theorem
above remains an independent demonstration of a different hard resource,
namely growing noncommutative order memory.

\section{Relative approximation on the same sparse path}

Exact hardness does not by itself imply approximation hardness. In fact,
Jerrum--Sinclair--Vigoda give an FPRAS for the permanent of an entrywise
nonnegative matrix \cite{JerrumSinclairVigoda2004}. The sparse-path identity,
however, also admits a different promise class.

\begin{theorem}[Real-PSD relative-norm transfer]
Unless $\mathrm{RP}=\mathrm{NP}$, there is no randomized polynomial-time
relative approximation scheme for the squared norm of every rational real
upper-bidiagonal matrix-pair input, polynomial jointly in the state dimensions,
coefficient bit length, inverse relative tolerance, and logarithmic inverse
failure probability.
\end{theorem}

\begin{proof}
Let $A\in\mathbb Q^{n\times n}$ be real positive semidefinite and apply the
sparse-path construction. Its squared norm is
\[
 s_A=\frac{\operatorname{perm}(A)^2}{(n!)^2}.
\]
The permanent of a real PSD matrix is nonnegative. If, with the prescribed
success probability,
\[
 (1-\epsilon)s_A\le \widetilde s_A\le(1+\epsilon)s_A,
\]
then $n!\sqrt{\widetilde s_A}$ estimates $\operatorname{perm}(A)$ between
factors $\sqrt{1-\epsilon}$ and $\sqrt{1+\epsilon}$. Choosing $\epsilon$
polynomially from a requested permanent tolerance gives a PRAS. Meiburg proves
that no PRAS exists for real PSD permanents unless
$\mathrm{RP}=\mathrm{NP}$ \cite{Meiburg2023}. The construction and the bit
length of $n!$ are polynomial, giving the claim.
\end{proof}

This theorem does not exclude additive estimates. Gurvits's estimator applies
to arbitrary complex matrices with additive scale
$\epsilon\lVert A\rVert^n$ \cite{AaronsonHance2014}. There is no uniform
conversion to relative exterior norm: for
\[
 A_\tau=\begin{pmatrix}1&1\\1&-1+\tau\end{pmatrix},
 \qquad \operatorname{perm}(A_\tau)=\tau,
\]
the coefficient scale remains of order one while the normalized exterior
squared norm is $\tau^2/4$. With $\tau=2^{-L}$, the required additive
tolerance is exponential in the input precision $L$.

\begin{lemma}[Rayleigh-quotient error certificate]
Let $n>0$, $h\in\mathbb R$, $E=h/n$, and suppose estimates satisfy
\[
 |n-\widetilde n|\le\Delta_n,\qquad
 |h-\widetilde h|\le\Delta_h,\qquad
 \widetilde n>\Delta_n.
\]
Writing $\widetilde E=\widetilde h/\widetilde n$, one has
\[
 |E-\widetilde E|
 \le
 \frac{\Delta_h+|\widetilde E|\Delta_n}
      {\widetilde n-\Delta_n}.
\]
For probabilistic bounds, the same statement holds on their simultaneous
success event and therefore with the union-bound confidence.
\end{lemma}

\begin{proof}
Use
\[
 E-\widetilde E
 =\frac{(h-\widetilde h)-\widetilde E(n-\widetilde n)}{n}
\]
and $n\ge\widetilde n-\Delta_n>0$.
\end{proof}

\paragraph{Scope.}
The fixed-bond, homogeneous pair-power, and sparse-path theorems concern exact
arithmetic. The PSD transfer separately rules out a generic relative scheme
under its stated complexity assumption. The sparse exact theorem rules out a generic
algorithm for a family that contains the upper-bidiagonal specialization; it
does not make every sparse instance hard. Neither result excludes other
structured restrictions, entrywise-nonnegative or additive promised
contractions, or ordered-sector first-quantized representations. None asserts
an unconditional separation of complexity classes.
